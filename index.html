PERSISTENCIA (GUARDAR Y CARGAR) window.onload = function() { actualizarSelectores(); loadWorkState(); // Recupera la jornada al refrescar }; function saveWorkState() { const state = { sesiones: DB.sesiones, currentIdx: DB.currentIdx, avance: DB.avance, staff: DB.staff, config: DB.config, activeView: DB.activeView }; localStorage.setItem('v86_work_state', JSON.stringify(state)); } function loadWorkState() { const saved = JSON.parse(localStorage.getItem('v86_work_state')); if (saved && saved.sesiones && saved.sesiones.length > 0) { DB.sesiones = saved.sesiones; DB.currentIdx = saved.currentIdx; DB.avance = saved.avance; DB.staff = saved.staff; DB.config = saved.config; DB.activeView = saved.activeView; cerrarModal('viewSetup'); document.getElementById(DB.activeView).classList.add('open'); document.getElementById('disp_sups').innerText = "SUPS: " + DB.config.sups.map(s => s.nom).join(', '); renderTabs(); renderSession(); renderChecklist(); } }
